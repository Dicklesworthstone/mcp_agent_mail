{# SQLite FTS snippets contain untrusted message text, not sanitized HTML. Only exact, balanced highlight markers become markup. Force-escape every text segment, even if a caller accidentally supplies a Markup value. #} {% macro highlighted_snippet(snippet) -%} {%- for part in snippet.split('') -%} {%- if loop.first -%} {{- part|forceescape -}} {%- else -%} {%- set highlight = part.split('', 1) -%} {%- if highlight|length == 2 -%} {{ highlight[0]|forceescape }}{{ highlight[1]|forceescape }} {%- else -%} {{- ''|forceescape -}}{{- part|forceescape -}} {%- endif -%} {%- endif -%} {%- endfor -%} {%- endmacro %}
